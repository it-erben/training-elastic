# ============================================================
# ILM Setup für smarttransactions — Vollständiges Beispiel
# ============================================================
#
# Diese Befehle in Kibana Dev Tools ausführen, in genau
# dieser Reihenfolge. Am Ende hast du einen funktionierenden
# ILM-gesteuerten Index mit Rollover.
#
# Architektur:
#   - Hot:    0–30 Tage, aktive Indexierung + Suche
#   - Warm:  30–180 Tage, nur Lesen, komprimiert
#   - Cold: 180–365 Tage, selten abgefragt
#   - Delete: nach 365 Tagen automatisch gelöscht


# ============================================================
# SCHRITT 1: ILM Policy erstellen
# ============================================================
#
# Diese Policy definiert den gesamten Lebenszyklus eines
# Index — wann er überwechselt, wann er komprimiert wird,
# und wann er gelöscht wird.

PUT _ilm/policy/smarttransactions-lifecycle
{
  "policy": {
    "phases": {

      "hot": {
        "min_age": "0ms",
        "actions": {
          "rollover": {
            "max_age": "30d",
            "max_primary_shard_size": "50gb"
          },
          "set_priority": {
            "priority": 100
          }
        }
      },

      "warm": {
        "min_age": "30d",
        "actions": {
          "shrink": {
            "number_of_shards": 1
          },
          "forcemerge": {
            "max_num_segments": 1
          },
          "set_priority": {
            "priority": 50
          }
        }
      },

      "cold": {
        "min_age": "180d",
        "actions": {
          "set_priority": {
            "priority": 0
          },
          "readonly": {}
        }
      },

      "delete": {
        "min_age": "365d",
        "actions": {
          "delete": {}
        }
      }
    }
  }
}

# Prüfen, ob die Policy angelegt wurde:
GET _ilm/policy/smarttransactions-lifecycle
