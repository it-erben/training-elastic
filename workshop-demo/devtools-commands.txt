# ============================================================
# DEMO 1: text vs. keyword — Warum der Feldtyp alles ändert
# ============================================================
#
# → "Wir fangen mit einem einfachen Beispiel an.
#    Im generalstores-Index ist das Feld 'category' als text
#    gemappt. Schauen wir mal, was passiert, wenn wir darauf
#    eine Aggregation machen — also 'Top Kategorien'."

GET demo-generalstores/_search
{
  "size": 0,
  "aggs": {
    "kategorien": {
      "terms": { "field": "category" }
    }
  }
}

# → "Ihr seht: Elasticsearch weigert sich, oder liefert
#    tokenisierte Fragmente — 'lebens', 'mittel' statt
#    'Lebensmittel'. Das ist genau das Antipattern aus
#    unserem Handout: Ein text-Feld zerlegt den Wert in
#    einzelne Tokens. Für Aggregationen ist das unbrauchbar."
#
# → "Jetzt dasselbe auf dem optimierten Index, wo category
#    als keyword gemappt ist:"

GET demo-generalstores-optimized/_search
{
  "size": 0,
  "aggs": {
    "kategorien": {
      "terms": { "field": "category" }
    }
  }
}

# → "Saubere Kategorienamen. Genau das, was ihr in einem
#    Kibana-Dashboard sehen wollt. Eine einzige Zeile im
#    Mapping macht den Unterschied."


# ============================================================
# DEMO 2: enabled:false — Felder raus, Daten bleiben
# ============================================================
#
# → "Die größte Angst bei Mapping-Änderungen ist: Verliere
#    ich Daten? Schauen wir uns das an den Receipt-Daten an.
#    Im Original-Index sind die Kassenbon-Felder indexiert.
#    Ich kann danach suchen:"

GET demo-smarttransactions/_search
{
  "size": 1,
  "query": {
    "exists": { "field": "receipt.value.name" }
  }
}

# → "Treffer. Das Feld existiert im Index.
#    Jetzt der optimierte Index — dort ist receipt auf
#    enabled:false gesetzt:"

GET demo-smarttransactions-optimized/_search
{
  "size": 1,
  "query": {
    "exists": { "field": "receipt.value.name" }
  }
}

# → "Null Treffer. Das Feld ist nicht im Index.
#    Aber jetzt die entscheidende Frage — sind die
#    Daten weg? Schauen wir uns das Dokument direkt an:"

GET demo-smarttransactions-optimized/_doc/STX_000001

# → "Da ist es — das komplette Receipt-Objekt, alle Zeilen
#    des Kassenbons. Die Daten liegen in _source, sie sind
#    vollständig da. Eure Anwendung kann den Bon weiterhin
#    anzeigen. Ihr könnt nur nicht mehr in Kibana danach
#    suchen — und das wollte ohnehin niemand.
#    Dafür spart ihr ca. 40 Felder im Mapping."


# ============================================================
# DEMO 3: filter vs. must — Scoring bei Transaktionen
# ============================================================
#
# → "Aus der Schulung kennt ihr KQL. Unter der Haube
#    erzeugt Kibana daraus Bool-Queries. Da gibt es einen
#    wichtigen Unterschied: must berechnet einen Relevanz-
#    Score, filter nicht. Schauen wir uns das an."
#
# → "Erst mit must — berechnet einen Score:"

GET demo-smarttransactions/_search
{
  "size": 3,
  "_source": ["id", "status", "basket_info.sum"],
  "query": {
    "bool": {
      "must": [
        { "term": { "status": "ok" } },
        { "range": { "basket_info.sum": { "gte": 5000 } } }
      ]
    }
  }
}

# → "Schaut euch _score an — jeder Treffer hat einen Wert
#    größer als Null. Elasticsearch hat berechnet, wie
#    'relevant' diese Transaktion ist. Aber: Wer von euch
#    hat jemals nach der relevantesten Transaktion gesucht?"
#
# → "Jetzt mit filter — kein Score:"

GET demo-smarttransactions/_search
{
  "size": 3,
  "_source": ["id", "status", "basket_info.sum"],
  "query": {
    "bool": {
      "filter": [
        { "term": { "status": "ok" } },
        { "range": { "basket_info.sum": { "gte": 5000 } } }
      ]
    }
  }
}

# → "_score ist überall 0.0. Keine Berechnung, dafür
#    wird das Ergebnis gecacht. Bei Finanztransaktionen
#    ist filter praktisch immer der richtige Weg.
#    Die gute Nachricht: Kibana/KQL macht das bereits
#    automatisch. Aufpassen müsst ihr nur, wenn ihr
#    direkt gegen die API programmiert."


# ============================================================
# DEMO 4: Strukturvergleich — Feldanzahl und Indexgröße
# ============================================================
#
# → "Zum Schluss der Gesamtüberblick. Schauen wir uns an,
#    was die Optimierungen insgesamt bewirken:"

GET _cat/indices/demo-*?v&h=index,docs.count,store.size&s=index

# → "Gleiche Daten, gleiches JSON — aber der optimierte
#    smarttransactions-Index ist deutlich kleiner. Und
#    das bei nur 200 Dokumenten. Bei euren Millionen
#    von Transaktionen reden wir über einen erheblichen
#    Unterschied in Speicher und Indexierungs-Performance."
#
# → "Schauen wir noch, wie viele Felder die Indices haben:"

GET demo-smarttransactions/_mapping?filter_path=**.properties
GET demo-smarttransactions-optimized/_mapping?filter_path=**.properties

# → "Das Original hat über 150 Felder, der optimierte
#    Index gut 100. In eurem echten Mapping mit 513 Feldern
#    wäre die Einsparung noch drastischer — wir schätzen
#    ungefähr 230 Felder weniger."
#
# → "Jedes dieser Felder verbraucht Ressourcen im Cluster-
#    State, auf dem Heap, und bei jeder Indexierung. Weniger
#    Felder heißt: schnellere Indexierung, weniger Speicher,
#    stabilerer Cluster."


# ============================================================
# BONUS (falls Zeit übrig): object-Felder
# ============================================================
#
# → "Eine kurze Sache noch: Schaut euch mal an, welche
#    Werte das Feld merchant.object hat:"

GET demo-smarttransactions/_search
{
  "size": 0,
  "aggs": {
    "object_werte": {
      "terms": { "field": "merchant.object" }
    }
  }
}

# → "Immer derselbe Wert — 'general.merchants'. Das ist
#    der API-Ressourcentyp. Danach filtert niemand.
#    Trotzdem legt Elasticsearch dafür einen Index UND
#    doc_values an. Mit index:false und doc_values:false
#    verschwindet dieser Overhead komplett — und das Feld
#    zieht sich durch jede einzelne Referenz in eurem
#    Mapping. Das sind geschätzt 20 Felder, die man so
#    aus dem Index entfernen kann."
